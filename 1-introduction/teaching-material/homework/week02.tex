\begin{core}[Shape Gallery (Scratch, about 10 minutes)]
Draw a triangle, a square and a hexagon in a row, like pictures in a gallery.
\begin{enumerate}[leftmargin=1.4em,itemsep=1pt]
  \item Open Scratch (\menu{File} $\rightarrow$ \menu{New} if an old project is open). Click \menu{Add Extension} (bottom left) and choose \menu{Pen}. Set the cat's \menu{Size} to \texttt{40}.
  \item Build the script. \enquote{pen up} before each \enquote{go to} stops the cat drawing a line between the shapes.
  \item Fill in the two missing numbers for the hexagon (use the rule \textbf{turn = 360 ÷ sides}).
  \item Click the flag. Save as \texttt{shape-gallery}.
\end{enumerate}
\begin{center}
\begin{scratch}[scale=0.72]
\blockinit{when \greenflag clicked}
\blockpen{erase all}
\blockmove{point in direction \ovalnum{90}}

\begin{challenge}[The circle trick]
Draw a shape with \enquote{repeat 36}, \enquote{move 10 steps}, \enquote{turn 10 degrees}. What does it look like? Why do you think it looks like that? (Hint: 36 × 10 = ?)
\answerlines{2}
What turn would you need for a shape with 10 sides? \rule{2cm}{0.4pt}
\end{challenge}
