% Junior Level 3, Week 9 -- play-at-home card
\homeworkweek{9}{Project Studio}{Optional. One short game with a grown-up.}

\begin{playathome}[Practice the show]
Next week is the graduation showcase. Practice with a grown-up.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Hold up your plan card. Say your three parts in three sentences.
  \item Point at one part. Say what the code does: \enquote{This part \ldots}
  \item Say one thing you changed to make it yours.
\end{enumerate}
\end{playathome}

\begin{playathome}[The squint test]
Draw your project with crayons. Put it across the room. Squint your eyes. Can you still see every part? If not, make the lines thicker or the colors darker. Write one idea to try next week.
\end{playathome}

\begin{selfcheck}
  \item I said my three parts in three sentences.
  \item I can say what one part of my code does.
  \item My project is easy for everyone to see.
\end{selfcheck}
