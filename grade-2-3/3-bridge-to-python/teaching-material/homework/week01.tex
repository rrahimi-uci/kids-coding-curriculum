% Junior Level 3, Week 1 -- play-at-home card
\homeworkweek{1}{Patterns in Patterns}{Optional. One short game with a grown-up.}

\begin{playathome}[Pattern maker]
Make a pattern for your grown-up to perform.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Pick two moves, like clap and pat your knees.
  \item Make a small pattern: clap, clap, pat. Do it 3 times.
  \item Now add a big pattern around it: the small pattern, then a jump. Do all of that 2 times.
  \item Your grown-up performs it. Did they get it right? Count the jumps together.
  \item Swap: your grown-up makes one for you.
\end{enumerate}
\end{playathome}

\begin{playathome}[Draw a nested pattern]
On paper, draw a row: circle, circle, triangle. Repeat the row 3 times. Then draw a star. Do all of that 2 times. Color it in your own way.
\end{playathome}

\begin{selfcheck}
  \item I made a pattern inside a pattern.
  \item I can say what a nested repeat is.
\end{selfcheck}
