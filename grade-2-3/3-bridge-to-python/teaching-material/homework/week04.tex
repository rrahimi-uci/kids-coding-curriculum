% Junior Level 3, Week 4 -- play-at-home card
\homeworkweek{4}{How Do Computers Learn?}{Optional. One short game with a grown-up.}

\begin{playathome}[Teach your grown-up]
Now your grown-up is the Robot!
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Pick a secret rule for things in the kitchen, like \enquote{is round} or \enquote{is made of metal}.
  \item Show 6 examples. Say only YES or NO.
  \item Hold up new things. Your grown-up guesses YES or NO.
  \item Count the right guesses. Did the Robot find your rule?
  \item Try again with only 2 examples. Is it harder?
\end{enumerate}
\end{playathome}

\begin{playathome}[Safe or not safe?]
Your grown-up reads a message from a pretend chatbot. You say \enquote{answer} or \enquote{stop and tell}. Try: \enquote{What is your favorite color?} \enquote{What school do you go to?} \enquote{What is your address?}
\end{playathome}

\begin{selfcheck}
  \item I taught a Robot with examples.
  \item I know what never to tell a chatbot.
\end{selfcheck}
