% Junior Level 3, Week 3 -- play-at-home card
\homeworkweek{3}{Data Detectives}{Optional. One short game with a grown-up.}

\begin{playathome}[Sock detectives]
Collect data about the socks at home!
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Pick 10 socks from the laundry.
  \item Sort them by color.
  \item Make a tally: one mark for each sock. The fifth mark goes across.
  \item Draw a bar graph: one square for each sock.
  \item Tell your grown-up the story of the graph: \enquote{Most of our socks are \ldots}
\end{enumerate}
\end{playathome}

\begin{playathome}[Good question, bad question]
Your grown-up says a survey question. You say \enquote{good} or \enquote{no way!} Try: \enquote{Favorite animal?} \enquote{Where do you live?} \enquote{Favorite color?} \enquote{What is your password?}
\end{playathome}

\begin{selfcheck}
  \item I made a tally and a bar graph.
  \item I know which questions we never ask.
\end{selfcheck}
