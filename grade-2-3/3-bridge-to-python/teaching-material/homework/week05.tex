% Junior Level 3, Week 5 -- play-at-home card
\homeworkweek{5}{Play Day: Blocks to Words}{Optional. One short game with a grown-up.}

\begin{playathome}[Be the Python turtle]
Your grown-up reads a Python line. You do it!
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item \texttt{t.forward(3)}: take three big steps.
  \item \texttt{t.left(90)}: make a quarter turn left.
  \item \texttt{t.right(180)}: make a half turn.
  \item \texttt{for i in range(4):} with \texttt{t.forward(2)} and \texttt{t.left(90)} inside: walk a square!
\end{enumerate}
Did you end where you started? Now swap. You read, your grown-up walks.
\end{playathome}

\begin{playathome}[Spot the twin]
Tell your grown-up the Scratch twin of each line: \texttt{t.forward(100)}, \texttt{t.left(90)}, \texttt{for i in range(4):}, \texttt{t.color("blue")}.
\end{playathome}

\begin{selfcheck}
  \item I walked a square from Python words.
  \item I can say the Scratch twin of a Python line.
  \item I told my grown-up one thing I am proud of.
\end{selfcheck}
