\chapterweek{2}{Plan It on Paper}{Big programs are small parts, planned first.}

\section{What you will learn}
\begin{ican}
  \item I can plan a program in parts on paper.
  \item I can test each part on its own.
  \item I can fix parts that do not fit together.
\end{ican}

\section{Plan first, then build}
A big program is made of small \textbf{parts}. Each part has a short list of steps. Build one part. Test it. Then build the next part.

\begin{exercise}[Party Time]
Build the plan one part at a time.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Make a variable called \enquote{pops}. Build Part 1 for the cat.
  \item Test Part 1. Check the tested box.
  \item Add Balloon1. Build Part 2 and Part 3 for the balloon.
  \item Test each part. Click a script to run it.
  \item Click the flag. Do all the parts work together?
\end{enumerate}
\begin{center}
\begin{scratch}[scale=0.75]
\blockinit{when \greenflag clicked}
\blockvariable{set \selectmenu{pops} to \ovalnum{0}}
\blocklook{say \ovalnum{Party time!} for \ovalnum{2} seconds}
\blockevent{broadcast \selectmenu{balloons}}
\end{scratch}
\qquad
\begin{scratch}[scale=0.75]
\blockinit{when \greenflag clicked}
\blocklook{hide}
\end{scratch}

\begin{scratch}[scale=0.75]
\blockinit{when I receive \selectmenu{balloons}}
\blockmove{go to \selectmenu{random position}}
\blocklook{show}
\end{scratch}
\qquad
\begin{scratch}[scale=0.75]
\blockinit{when this sprite clicked}
\blocksound{start sound \selectmenu{Pop}}
\blockvariable{change \selectmenu{pops} by \ovalnum{1}}
\blockmove{go to \selectmenu{random position}}
\end{scratch}
\end{center}
\end{exercise}

\begin{exercise}[The Silent Party]
Leo's parts all pass their tests. But with the flag, the balloon never comes! Change your cat's message to Leo's message, \enquote{party}. Test it.
\begin{center}
\begin{scratch}[scale=0.75]

\end{scratch}
\end{center}
Circle the bug. Fix it on your Mac.
\end{exercise}

\begin{vocab}
  \vocabword{part}{A small piece of a big program. You can build and test it on its own.}
  \vocabword{step list}{The steps for one part, written in order.}
  \vocabword{test}{Run it and check that it does what the plan says.}
\end{vocab}
