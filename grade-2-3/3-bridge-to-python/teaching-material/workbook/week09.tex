\begin{exercise}[My Plan Card]
Fill in your plan card.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Circle A or B.
  \item Draw what it will look like.
  \item Name three parts, in order.
  \item Check the must-haves.
  \item Write whose idea you used.
\end{enumerate}
\end{exercise}

\begin{exercise}[Build My Big Build]
Build one part. Test it. Check it off. Then build the next part.

\textbf{Path A.} Leo's Flower Garden. Press space for a flower!
\begin{center}
\begin{scratch}[scale=0.75]
\blockinit{when \greenflag clicked}
\blockpen{erase all}
\blockvariable{set \selectmenu{flowers} to \ovalnum{0}}
\end{scratch}
\quad
\begin{scratch}[scale=0.75]
\blockinit{when \selectmenu{space} key pressed}
\blockmove{go to \selectmenu{mouse-pointer}}
\blockpen{pen down}
\blockrepeat{repeat \ovalnum{6}}{
  \blockrepeat{repeat \ovalnum{4}}{
    \blockmove{move \ovalnum{30} steps}
    \blockmove{turn \turnright{} \ovalnum{90} degrees}
  }
  \blockmove{turn \turnright{} \ovalnum{60} degrees}
}
\blockpen{pen up}
\blockvariable{change \selectmenu{flowers} by \ovalnum{1}}
\end{scratch}
\end{center}

\textbf{Path B.} Omar's postcard. He changed a color and a number. Press Run after every change!

\noindent\begin{minipage}{\linewidth}
\begin{pycode}
t.color("blue")
for i in range(4):
    t.forward(120)
    t.left(90)
\end{pycode}
\end{minipage}
\end{exercise}

\begin{exercise}[Partner Check]
Swap with a partner. Check their project. Circle yes or not yet.
\begin{center}
\begin{tabular}{@{}l c@{}}
It starts with the green flag or Run. & yes \quad not yet\\
Part 1 works. & yes \quad not yet\\
Part 2 works. & yes \quad not yet\\
Part 3 works. & yes \quad not yet\\
It has every must-have. & yes \quad not yet\\
Everyone can see it. & yes \quad not yet\\
\end{tabular}
\end{center}
Tell your partner one \enquote{not yet}: \enquote{I expected \ldots, but it \ldots}
\end{exercise}
