\kctitle{My First Arcade Game}{Project brief, game plan card and stars (weeks 9 and 10)}

\begin{grownup}[How to use this brief]
Read it together at the start of week 9. Your child fills in the game plan card before building.
Writing a few words, drawing and circling all count.
\end{grownup}

\section{Your mission}
Make your own arcade game in Scratch. Plan it first. Then build it.\par
Someone else plays it and finds a bug. You fix it.\par
In week 10, your family plays it at the arcade!

\begin{safety}[Nicknames, credit and fair play]
Put a nickname on your game, never a full name.
Say where your sprites and sounds come from: \enquote{The apple is from the Scratch library.}
Tell players the rules at the start, so the game is fair.
\end{safety}

\section{Need a different path?}
\begin{tip}[Core path: start from a template]
\begin{minipage}[t]{0.52\linewidth}\RaggedRight
A grown-up gives you a game to start from.\par\smallskip
The player already moves with the arrow keys.\par\smallskip
Add one must-have at a time. Play it after each one.
\end{minipage}\hfill
\begin{minipage}[t]{0.44\linewidth}\vspace{0pt}
\begin{scratch}[scale=0.62]
\blockinit{when \selectmenu{right arrow} key pressed}
\blockmove{point in direction \ovalnum{90}}
\blockmove{move \ovalnum{10} steps}
\end{scratch}
\end{minipage}
\end{tip}

\begin{stretchchallenge}[Stretch path: a My Block or a second level]
\begin{minipage}[t]{0.5\linewidth}\RaggedRight
\textbf{A My Block:} give a group of blocks a name. Use it in two places.\par\smallskip
This block sends the apple back to the top. It hops right a surprise number of times.\par\smallskip
\textbf{Or a second level:} when the score is 10, change the backdrop and make the apple faster!
\end{minipage}\hfill
\begin{minipage}[t]{0.46\linewidth}\vspace{0pt}
\begin{scratch}[scale=0.58]
\initmoreblocks{define \namemoreblocks{new apple}}
\blockmove{go to \selectmenu{Star}}
\blockmove{point in direction \ovalnum{90}}
\blockrepeat{repeat \ovaloperator{pick random \ovalnum{1} to \ovalnum{10}}}{
  \blockmove{move \ovalnum{30} steps}
}
\blockmove{point in direction \ovalnum{180}}
\end{scratch}
\end{minipage}
\end{stretchchallenge}
