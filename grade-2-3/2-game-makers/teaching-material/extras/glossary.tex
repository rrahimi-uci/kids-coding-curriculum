\kctitle{Word Book}{Every new word in Game Makers, from A to Z}

\begin{grownup}[How to use the word book]
Read a word and its meaning aloud, then ask your child to show you the word on the computer or act
it out. The gray label says which week the word first comes up (W1 = week 1).
\end{grownup}

\gl{arcade}{W10}{A place with lots of games to play.}

\letter{B}
\gl{bounce}{W1}{Turn around at the edge and go back.}
\gl{broadcast}{W6}{Send a message to every sprite at once.}

\letter{C}
\gl{change}{W3}{Add to the number that is already in the box.}
\gl{credit}{W6}{Saying who made something, like the Scratch artists who drew the cat.}

\letter{D}
\gl{define}{W7}{The hat block on top of a My Block. It holds the blocks your block does.}
\gl{design choice}{W10}{Something you decide on purpose when you make a game, like how many misses.}

\letter{F}
\gl{forever}{W1}{A loop that never stops by itself. You click stop.}

\letter{G}
\gl{game over}{W8}{The ending when you lose. The game stops.}
\gl{goal}{W8}{What you try to do to win, like catch 10 apples.}
\gl{good sport}{W4}{Someone who says \enquote{good game} when they win and when they lose.}

\letter{I}
\gl{if}{W2}{A block that asks a question. It runs the blocks inside only when the answer is yes.}

\letter{M}
\gl{message}{W6}{A note one sprite sends to the others.}
\gl{My Block}{W7}{A block you make yourself. Its name stands for a group of blocks.}

\letter{P}
\gl{play day}{W5}{A day to try everything we know and show what we can do.}
\gl{playtest}{W9}{Someone plays your game so you can find bugs.}
\gl{practice}{W5}{Doing something again and again to get better at it.}
\gl{prototype}{W9}{A quick test version of a game, made before the real one.}

\letter{R}
\gl{random}{W4}{A surprise: nobody can guess what comes next, like a die.}

\letter{S}
\gl{set}{W3}{Put a new number in the box.}

\letter{T}
\gl{touching}{W2}{A question block: is my sprite touching that thing?}

\letter{V}
\gl{variable}{W3}{A box with a label that remembers a number, like score.}
