\section{What you will learn}
\begin{ican}
  \item I can be the computer and keep the score.
  \item I can find a bug in a game and fix it.
  \item I can make a game and say how it works.
\end{ican}

\begin{exercise}[Be the Computer]
You are the cat. Do the script one block at a time. Each 10 steps is one square. What is the score at the end?
\begin{center}
\begin{scratch}[scale=0.8]
\blockinit{when \greenflag clicked}
\blockvariable{set \selectmenu{score} to \ovalnum{0}}
\blockrepeat{repeat \ovalnum{6}}{
  \blockmove{move \ovalnum{10} steps}
  \blockif{if \boolsensing{touching \ovalsensing*{Apple} ?} then}{
    \blockvariable{change \selectmenu{score} by \ovalnum{1}}
  }
}
\end{scratch}

\begin{exercise}[Bug-Hunt Relay]
Three games are broken. Build each one. Test it. Find the bug. Fix it. Get a sticker!

\textbf{Leg 1: Stuck at the Edge.} The cat should walk back and forth forever.
\begin{center}
\begin{scratch}[scale=0.8]
\blockinit{when \greenflag clicked}
\blockmove{set rotation style \selectmenu{left-right}}
\blockmove{if on edge, bounce}
\blockinfloop{forever}{
  \blockmove{move \ovalnum{10} steps}
}
\end{scratch}
\end{center}

\textbf{Leg 2: The Stuck Score.} Each click on the Ghost should add 1 to the score.
\begin{center}
\begin{scratch}[scale=0.8]
\blockinit{when \greenflag clicked}
\blockvariable{set \selectmenu{score} to \ovalnum{0}}
\end{scratch}
\qquad
\begin{scratch}[scale=0.8]
\blockinit{when this sprite clicked}
\blockvariable{set \selectmenu{score} to \ovalnum{1}}
\blocksound{start sound \selectmenu{Pop}}
\end{scratch}
\end{center}

\textbf{Leg 3: The Apple That Won't Pop.} The cat follows the mouse. When it touches the Apple, the Apple should pop and jump away.
\begin{center}
\begin{minipage}[t]{0.4\linewidth}\centering
\textbf{Cat's code}\par\smallskip
\begin{scratch}[scale=0.8]
\blockinit{when \greenflag clicked}
\blockinfloop{forever}{
  \blockmove{go to \selectmenu{mouse-pointer}}
}
\end{scratch}
\end{minipage}\hfill
\begin{minipage}[t]{0.55\linewidth}\centering
\textbf{Apple's code}\par\smallskip
\begin{scratch}[scale=0.8]
\blockinit{when \greenflag clicked}
\blockinfloop{forever}{
  \blockif{if \boolsensing{touching \ovalsensing*{edge} ?} then}{
    \blocksound{start sound \selectmenu{Pop}}
    \blockmove{go to \selectmenu{random position}}
  }
}
\end{scratch}
\end{minipage}
\end{center}
\end{exercise}

\begin{exercise}[My Best Game]
Keep leg 3 open. Make it a real game!
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Make a variable called \enquote{score}.
  \item In the Apple's code, set the score to 0 at the start.
  \item Add 1 to the score each time the cat catches the Apple.
  \item Add your own idea: a sound, a backdrop, a new sprite.
\end{enumerate}
Tell your grown-up how your game works.
\end{exercise}
