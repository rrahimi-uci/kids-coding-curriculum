\chapterweek{2}{If Touching\ldots}{If-then lets a sprite check something and choose.}

\section{What you will learn}
\begin{ican}
  \item I can make a sprite check \enquote{if touching}.
  \item I can build a catch game.
  \item I can move an if into a forever loop.
\end{ican}

\section{Ask and choose}
An \textbf{if} asks a question. Yes? It does the blocks inside. No? It skips them. Put the if inside a forever loop. Then it asks again and again.

\needspace{14\baselineskip}
\begin{exercise}[Boo Ghost]
Add the Ghost. Click the flag. Touch the ghost with the mouse!
\begin{center}
\begin{scratch}[scale=0.85]
\blockinit{when \greenflag clicked}
\blockinfloop{forever}{
  \blockif{if \boolsensing{touching \ovalsensing*{mouse-pointer} ?} then}{
    \blocklook{say \ovalnum{Boo!} for \ovalnum{1} seconds}
  }
}
\end{scratch}
\end{center}
\end{exercise}

\begin{exercise}[Catch the Apple]
Start a new project. Add the Star, the Bowl and the Apple.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Drag the Star near the top. Give it the hide script.
  \item Drag the Bowl to the bottom. Build its three scripts.
  \item Build the Apple's script. The two ifs go inside the forever.
  \item Click the flag. Catch apples with the arrow keys!
\end{enumerate}
\begin{center}
\begin{minipage}[t]{0.4\linewidth}
\textbf{Star:}\par\smallskip
\begin{scratch}[scale=0.7]
\blockinit{when \greenflag clicked}
\blocklook{hide}
\end{scratch}

\medskip
\textbf{Bowl:}\par\smallskip
\begin{scratch}[scale=0.7]
\blockinit{when \greenflag clicked}
\blockmove{set rotation style \selectmenu{left-right}}
\end{scratch}

\smallskip
\begin{scratch}[scale=0.7]
\blockinit{when \selectmenu{right arrow} key pressed}
\blockmove{point in direction \ovalnum{90}}
\blockmove{move \ovalnum{10} steps}
\end{scratch}

\smallskip
\begin{scratch}[scale=0.7]
\blockinit{when \selectmenu{left arrow} key pressed}
\blockmove{point in direction \ovalnum{270}}
\blockmove{move \ovalnum{10} steps}
\end{scratch}
\end{minipage}\hfill
\begin{minipage}[t]{0.56\linewidth}
\textbf{Apple:}\par\smallskip
\begin{scratch}[scale=0.7]
\blockinit{when \greenflag clicked}
\blockmove{set rotation style \selectmenu{left-right}}
\blockmove{go to \selectmenu{Star}}
\blockmove{point in direction \ovalnum{180}}
\blockinfloop{forever}{
  \blockmove{move \ovalnum{5} steps}
  \blockif{if \boolsensing{touching \ovalsensing*{Bowl} ?} then}{
    \blocksound{start sound \selectmenu{Pop}}
    \blockmove{go to \selectmenu{Star}}
  }
  \blockif{if \boolsensing{touching \ovalsensing*{edge} ?} then}{
    \blockmove{go to \selectmenu{Star}}
  }
}
\end{scratch}
\end{minipage}
\end{center}
\end{exercise}

\begin{exercise}[Ghost Tag]
Kai's ghost should say \enquote{Got you!} when it touches the Ballerina. It flies right through her!
\begin{center}
\begin{scratch}[scale=0.75]
\blockinit{when \greenflag clicked}
\blockmove{point in direction \ovalnum{90}}
\blockmove{set rotation style \selectmenu{left-right}}
\blockif{if \boolsensing{touching \ovalsensing*{Ballerina} ?} then}{
  \blocklook{say \ovalnum{Got you!} for \ovalnum{1} seconds}
}
\blockinfloop{forever}{
  \blockmove{move \ovalnum{5} steps}
  \blockmove{if on edge, bounce}
}
\end{scratch}
\end{center}
Circle the if that checks only once. Fix it on your computer.
\end{exercise}

\begin{vocab}
  \vocabword{if}{A block that asks a question. It runs the blocks inside only when the answer is yes.}
  \vocabword{touching}{A question block: is my sprite touching that thing?}
\end{vocab}
