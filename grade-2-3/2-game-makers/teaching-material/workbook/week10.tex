\chapterweek{10}{Arcade Showcase}{Show, explain and celebrate.}

\section{What you will learn}
\begin{ican}
  \item I can show my game in one minute.
  \item I can say one choice I made, and why.
  \item I can give kind sticker feedback.
\end{ican}

\section{Arcade day}
Today our home is an arcade! You show your game. A Player plays it. Then we celebrate!

\begin{exercise}[Last-Minute Bug Check]
Check your game with a partner. Circle yes or not yet.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Click the flag. Does it start at 0? \hfill yes \quad not yet
  \item Play to win. Does it say You win! and stop? \hfill yes \quad not yet
  \item Click the flag again. Play to lose. Game over? \hfill yes \quad not yet
  \item Can you hear the sound? \hfill yes \quad not yet
\end{enumerate}
Fix one thing. Then save.
\end{exercise}

\begin{exercise}[My One-Minute Demo]
Mei points at her referee script:
\begin{center}
\begin{scratch}[scale=0.8]
\blockif{if \booloperator{\ovalvariable{misses} = \ovalnum{3}} then}{
  \blocklook{say \ovalnum{Game over!} for \ovalnum{2} seconds}
  \blockstop{stop \selectmenu{all}}
}
\end{scratch}
\end{center}
Mei says: \enquote{I chose 3 misses, because my Player said 1 was too hard.}

Now you:
\begin{itemize}[leftmargin=1.4em,itemsep=6pt]
  \item My game is called \rule{5cm}{0.4pt}. You are the \rule{3cm}{0.4pt}.
  \item You win when \rule{6.5cm}{0.4pt}.
  \item I chose \rule{4cm}{0.4pt}, because \rule{4cm}{0.4pt}.
\end{itemize}
\end{exercise}
