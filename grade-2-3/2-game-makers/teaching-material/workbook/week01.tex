\section{Welcome, Game Maker!}
Games keep going. A \textbf{forever} loop runs the blocks inside it again and again. It stops only when you click the red stop sign.

\begin{exercise}[Bounce Cat]
Make the cat run from side to side, forever.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Yellow \menu{Events}: the green flag block.
  \item Blue \menu{Motion}: point in direction 90 and set rotation style.
  \item Orange \menu{Control}: forever.
  \item Blue \menu{Motion}: move and bounce go inside.
  \item Click the flag. Then click the stop sign.
\end{enumerate}
\begin{center}
\begin{scratch}[scale=0.85]
\blockinit{when \greenflag clicked}
\blockmove{point in direction \ovalnum{90}}
\blockmove{set rotation style \selectmenu{left-right}}
\blockinfloop{forever}{
  \blockmove{move \ovalnum{10} steps}
  \blockmove{if on edge, bounce}
}
\end{scratch}
\end{center}
\end{exercise}

\needspace{16\baselineskip}
\begin{exercise}[Dance Forever]
Add the Ballerina. Build both scripts.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Press space. Count her dance moves.
  \item Click the flag. She dances and dances.
  \item Click the stop sign. Which one stopped by itself?
\end{enumerate}
\begin{center}
\begin{scratch}[scale=0.8]
\blockinit{when \selectmenu{space} key pressed}
\blockrepeat{repeat \ovalnum{4}}{
  \blocklook{next costume}
  \blockcontrol{wait \ovalnum{1} seconds}
}
\end{scratch}
\qquad
\begin{scratch}[scale=0.8]
\blockinit{when \greenflag clicked}
\blockinfloop{forever}{
  \blocklook{next costume}
  \blockcontrol{wait \ovalnum{1} seconds}
}
\end{scratch}
\end{center}
\end{exercise}

\begin{exercise}[The Missing Hello]
Kai's cat should say \enquote{Watch me!}, meow, then bounce. It bounces, but it never talks!
\begin{center}
\begin{scratch}[scale=0.8]
\blockinit{when \greenflag clicked}
\blockinfloop{forever}{
  \blockmove{move \ovalnum{10} steps}
  \blockmove{if on edge, bounce}
}
\end{scratch}

\vspace{0.3cm}
\begin{scratch}[scale=0.8]
\blocklook{say \ovalnum{Watch me!} for \ovalnum{2} seconds}
\blocksound{start sound \selectmenu{Meow}}
\end{scratch}
\end{center}
Circle the blocks that never run. Fix it on your computer.
\end{exercise}
