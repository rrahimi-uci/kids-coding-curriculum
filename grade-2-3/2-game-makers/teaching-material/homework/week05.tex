% Junior Level 2, Week 5 -- play-at-home card
\homeworkweek{5}{Play Day and Show Me}{Optional. One short game with a grown-up.}

\begin{playathome}[Be the computer at home]
Put five cushions or sheets of paper in a row. Put a sock on two of them.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item The grown-up reads the script. You are the sprite.
  \item \enquote{Set score to 0.} Hold up a closed fist.
  \item \enquote{Repeat 5: step onto the next square. If you are touching a sock, change score by 1.} Put up one finger for each sock.
  \item What is the score at the end? Count your fingers!
\end{enumerate}
Now swap: you read, the grown-up is the sprite. Move the socks first!
\end{playathome}

\begin{playathome}[I got better at\ldots]
Tell your grown-up one thing you can do now that you could not do five weeks ago. Draw it on the back of this card.
\end{playathome}

\begin{selfcheck}
  \item I was the computer and kept the score.
  \item I told a grown-up how one of my games works.
  \item I said one thing I got better at.
\end{selfcheck}
