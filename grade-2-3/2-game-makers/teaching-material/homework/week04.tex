% Junior Level 2, Week 4 -- play-at-home card
\homeworkweek{4}{Surprise! Random}{Optional. One short game with a grown-up.}

\begin{playathome}[Dice race at home]
Draw a race track with 20 squares on paper. Draw a flag on the last square.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Each player picks a small toy as a racer.
  \item Take turns. Roll a die and hop that many squares. Count out loud.
  \item The first racer to reach the flag wins.
  \item Say \enquote{good game} and do a high five, whoever wins!
\end{enumerate}
\end{playathome}

\begin{playathome}[Can you guess it?]
The grown-up says a thing. You say \enquote{I can guess it!} or \enquote{Surprise!}
\begin{itemize}[leftmargin=1.4em,itemsep=2pt]
  \item What number the die shows next.
  \item What day comes after Monday.
  \item Which hand the grown-up hides a coin in.
  \item What color the sky is on a sunny day.
\end{itemize}
Which ones are random?
\end{playathome}

\begin{selfcheck}
  \item I played a dice race and said \enquote{good game}.
  \item I can say what random means.
  \item I found something random at home.
\end{selfcheck}
