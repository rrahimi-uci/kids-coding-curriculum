% Junior Level 2, Week 6 -- play-at-home card
\homeworkweek{6}{Messages}{Optional. One short game with a grown-up.}

\begin{playathome}[Pass the note at home]
Make three paper notes with a grown-up. Write one word inside each: CLAP, HOP and SING.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item The grown-up hands you the CLAP note. Clap, then pass the HOP note back.
  \item The grown-up hops, then passes you the SING note. Sing one line of a song.
  \item Now the tricky one: the grown-up passes a note that says DANCE. Nobody has a DANCE job. What happens?
\end{enumerate}
\end{playathome}

\begin{playathome}[Who made it?]
Pick a favorite picture book. Find the names of the people who made it: the writer and the artist. That is giving credit! Say thank you to them out loud.
\end{playathome}

\begin{selfcheck}
  \item I passed a note and waited for my message.
  \item I can say what \enquote{broadcast} means.
  \item I found who made my favorite book.
\end{selfcheck}
