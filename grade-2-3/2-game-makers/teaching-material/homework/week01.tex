% Junior Level 2, Week 1 -- play-at-home card
\homeworkweek{1}{Forever and Bounce}{Optional. One short game with a grown-up.}

\begin{playathome}[Forever freeze dance]
Pick a song with a grown-up.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item The music plays. You dance and dance, like a forever loop.
  \item The grown-up pauses the music. Freeze!
  \item Swap: you hold the pause button. The grown-up dances.
\end{enumerate}
Then try a bounce walk: walk across the room. Touch the wall, turn around, walk back.
\end{playathome}

\begin{playathome}[My stop plan]
Games and videos can go on forever. You are the stop sign! Make a stop plan with your grown-up and draw it: \enquote{When the timer rings, I will\ldots}
\end{playathome}

\begin{selfcheck}
  \item I danced until the music stopped.
  \item I can say how forever is different from repeat.
  \item I have a stop plan for when the timer rings.
\end{selfcheck}
