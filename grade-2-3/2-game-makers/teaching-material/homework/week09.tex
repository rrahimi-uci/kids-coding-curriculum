% Junior Level 2, Week 9 -- play-at-home card
\homeworkweek{9}{Project Studio}{Optional. One short game with a grown-up.}

\begin{playathome}[Practice the showcase]
Next week is the arcade showcase. Practice with a grown-up.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Say: \enquote{My game is called \ldots. You are the \ldots.}
  \item Say how to win and how to lose.
  \item Say one choice you made, and why: \enquote{I chose 3 misses, because \ldots}
\end{enumerate}
\end{playathome}

\begin{playathome}[Arcade poster]
Make a poster for your game. Write its name in big letters. Draw the player and the thing to catch or avoid. Add one line: how to win. Use shapes and colors, so everyone can read it.
\end{playathome}

\begin{selfcheck}
  \item My game plan card is filled in.
  \item A Player tested my game, and I fixed a bug.
  \item I ask before I show someone else's game.
\end{selfcheck}
