% Junior Level 2, Week 10 -- play-at-home card
\homeworkweek{10}{Arcade Showcase}{Optional. One short game with a grown-up.}

\begin{playathome}[Family arcade]
Show your game to someone in your family.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Say how to win and how to lose.
  \item Let them play. Hands in your lap!
  \item Ask them for two stars and a wish.
\end{enumerate}
\end{playathome}

\begin{playathome}[What next?]
Draw one game you want to make next. A second level? A new maze? Put your certificate somewhere you can see it!
\answerspace{3cm}
\end{playathome}

\begin{selfcheck}
  \item I showed my game to my family.
  \item I said one design choice, and why.
  \item I used my nickname, not my full name.
\end{selfcheck}
