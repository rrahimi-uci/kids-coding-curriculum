% Junior Level 1, Week 3 -- play-at-home card
\homeworkweek{3}{Bugs Are Clues}{Optional. One short game with a grown-up.}

\begin{playathome}[Spot the bug]
A grown-up is a robot brushing its teeth: toothpaste on the brush, brush, rinse.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item The robot acts it out, but makes one bug on purpose.
  \item Maybe a step is in the wrong order. Maybe it is too fast to see.
  \item Spot the bug. Say it kindly: \enquote{I think I found a clue!}
  \item Tell the robot how to fix it. Is it fixed?
  \item Swap: you are the robot, and your grown-up spots the bug.
\end{enumerate}
\end{playathome}

\begin{playathome}[Slow motion]
Do three jumps super fast. Now do them again with a wait: jump, count to one, jump, count to one, jump. Which way can a friend see better?
\end{playathome}

\begin{selfcheck}
  \item I found a bug and said it kindly.
  \item I know what a wait does.
\end{selfcheck}
