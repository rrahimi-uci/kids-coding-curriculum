% Junior Level 1, Week 2 -- play-at-home card
\homeworkweek{2}{Meet Scratch}{Optional. One short game with a grown-up.}

\begin{playathome}[Puppet show at home]
Make three cards with a grown-up: \enquote{SAY hello}, \enquote{MOVE 3 steps}, \enquote{MAKE a sound}.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Put the cards in a line. You read them. A toy is the sprite.
  \item The toy does each card, top to bottom.
  \item Swap the order of two cards. What changes?
\end{enumerate}
\end{playathome}

\begin{playathome}[If you have Scratch at home]
Show a grown-up how to make the cat say hi. Then change the words or the sound together.
\end{playathome}

\begin{selfcheck}
  \item I played the puppet show.
  \item I can say what a sprite is.
\end{selfcheck}
