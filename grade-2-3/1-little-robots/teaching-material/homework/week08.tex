% Junior Level 1, Week 8 -- play-at-home card
\homeworkweek{8}{Draw With the Pen}{Optional. One short game with a grown-up.}

\begin{playathome}[My greeting card idea]
Soon you will make a greeting card in Scratch. Who will your card be for?
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Pick someone special. Grandma? A friend? Your cat?
  \item Draw one idea for the card in the box.
  \item Tell a grown-up: what will happen in your card?
\end{enumerate}
\answerspace{4cm}
\end{playathome}

\begin{playathome}[Walk a square]
Be the drawing robot. A grown-up reads: \enquote{Repeat 4: walk 3 steps, quarter turn.}
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Walk it. Did you get back to the start?
  \item Now draw a square on paper. Say \enquote{move, quarter turn} for each side.
\end{enumerate}
\end{playathome}

\begin{selfcheck}
  \item I drew an idea for my card.
  \item I walked a square.
  \item I know a password is secret, like a toothbrush.
\end{selfcheck}
