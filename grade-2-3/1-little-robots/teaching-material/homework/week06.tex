% Junior Level 1, Week 6 -- play-at-home card
\homeworkweek{6}{When I Click\ldots}{Optional. One short game with a grown-up.}

\begin{playathome}[When I clap, you jump]
Make three event cards with a grown-up. Draw a picture on each one.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item \enquote{When I CLAP: jump.}
  \item \enquote{When I KNOCK: freeze.}
  \item \enquote{When I say PIZZA: sit down.}
\end{enumerate}
The grown-up makes the events. You do the actions. Then swap: you make the events and the grown-up does the actions. Try a tricky one: say \enquote{banana}. What happens?
\end{playathome}

\begin{playathome}[A nickname for my sprite]
Think of a fun nickname for your cat, like Captain Fluff. If you have Scratch at home, make the cat say it when you click it.
\end{playathome}

\begin{selfcheck}
  \item I played the event game.
  \item I can name three ways to start a script.
  \item My sprites use nicknames, never my real name.
\end{selfcheck}
