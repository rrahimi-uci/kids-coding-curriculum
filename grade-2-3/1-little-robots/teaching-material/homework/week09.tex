% Junior Level 1, Week 9 -- play-at-home card
\homeworkweek{9}{Project Studio}{Optional. One short game with a grown-up.}

\begin{playathome}[Practice the show]
Next week is the showcase. Practice with a grown-up.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Hold up your storyboard. Tell the story in three sentences: start, middle, end.
  \item Say: \enquote{My card is for \ldots}
  \item Say one thing your code does: \enquote{When you click the \ldots, it \ldots}
\end{enumerate}
\end{playathome}

\begin{playathome}[Finish a box]
Is one box of your storyboard not finished? Draw it now. Add color. If you have Scratch at home, build it together.
\end{playathome}

\begin{selfcheck}
  \item I told my card story in three sentences.
  \item I can say what one script does.
  \item I ask before I show someone else's work.
\end{selfcheck}
