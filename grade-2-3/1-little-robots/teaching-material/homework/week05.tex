% Junior Level 1, Week 5 -- play-at-home card
\homeworkweek{5}{Play Day and Show Me}{Optional. One short game with a grown-up.}

\begin{playathome}[Arrow maze at home]
Make a maze on the floor with a grown-up. Use floor tiles or sheets of paper as squares.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Put a toy on the goal square. Put a cushion on one square: that is a tree.
  \item Use your robot cards from week 1: FORWARD, TURN LEFT, TURN RIGHT.
  \item The grown-up is the robot. The robot does your cards in order.
  \item Did the robot reach the toy? If not, find the wrong card and fix it.
\end{enumerate}
\end{playathome}

\begin{playathome}[Spot the bug]
Now the grown-up lays the cards, with one bug on purpose. You are the robot. Do the cards. Can you find the wrong card?
\end{playathome}

\begin{selfcheck}
  \item I made a maze and the robot reached the goal.
  \item I found a bug and fixed it.
  \item I told a grown-up what one of my scripts does.
\end{selfcheck}
