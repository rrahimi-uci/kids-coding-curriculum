\begin{exercise}[Bug-Hunt Relay]
Three projects are broken. Build each one as you see it. Test it. Find the bug. Fix it. Get a sticker!

\textbf{Leg 1: The Quiet Cat.} It should say hi and meow.
\begin{center}
\begin{scratch}[scale=0.85]
\blockinit{when \greenflag clicked}
\blocklook{say \ovalnum{Hi!} for \ovalnum{2} seconds}
\end{scratch}

\vspace{0.3cm}
\begin{scratch}[scale=0.85]
\blocksound{play sound \selectmenu{Meow} until done}
\end{scratch}
\end{center}

\textbf{Leg 2: Bye Before Hi.} It should say hi first, then bye.
\begin{center}
\begin{scratch}[scale=0.85]
\blockinit{when \greenflag clicked}
\blocklook{say \ovalnum{Bye!} for \ovalnum{2} seconds}
\blocklook{say \ovalnum{Hi!} for \ovalnum{2} seconds}
\end{scratch}
\end{center}

\textbf{Leg 3: Too Many Meows.} It should meow three times.
\begin{center}
\begin{scratch}[scale=0.85]
\blockinit{when \greenflag clicked}
\blockrepeat{repeat \ovalnum{10}}{
  \blocksound{play sound \selectmenu{Meow} until done}
}
\end{scratch}
\end{center}
\end{exercise}

\begin{exercise}[My Best Show]
Make your own show. It needs the green flag and a repeat. Here is one idea.
\begin{center}
\begin{scratch}[scale=0.85]
\blockinit{when \greenflag clicked}
\blocklook{say \ovalnum{Watch me!} for \ovalnum{2} seconds}
\blockrepeat{repeat \ovalnum{4}}{
  \blocklook{next costume}
  \blockmove{move \ovalnum{10} steps}
  \blockcontrol{wait \ovalnum{1} seconds}
}
\blocklook{say \ovalnum{Ta-da!} for \ovalnum{2} seconds}
\end{scratch}
\end{center}
Change the words or the numbers. Add a sound if you like. Then tell your grown-up what your script does.
\end{exercise}

\end{grownup}
