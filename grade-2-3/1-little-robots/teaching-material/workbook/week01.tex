\section{Robot cards}
A robot does exactly what you tell it. It cannot guess.

FORWARD means one square ahead. A turn is a quarter turn, on the spot. The robot does not step when it turns.

A row of cards is a \textbf{program}. The robot does the cards in order, from left to right.

\begin{exercise}[Parent Robot]
Your grown-up is the robot!
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Put the star on a square of the floor grid.
  \item Lay your cards in a row.
  \item Read the cards out loud, one at a time.
  \item The robot does each card. Did it reach the star?
  \item Swap jobs and play again.
\end{enumerate}
\end{exercise}

\begin{exercise}[The Wrong Card]
Ada wants the robot to reach the star. One card is in the wrong place.
\begin{center}
%% alt: A 3 by 3 grid. A robot is in the bottom-left square, facing up. A star is in the top-right square. Next to the grid, five cards in a row: FORWARD, TURN RIGHT, FORWARD, FORWARD, FORWARD.
\begin{tikzpicture}[x=1cm,y=1cm,
  edge/.style={draw=kcblack,line width=0.7pt,rounded corners=2pt},
  arr/.style={line width=2.5pt,-{Triangle[length=6pt,width=9pt]}}]
  \draw[kcgray,line width=0.8pt] (0,0) grid[step=1.2] (3.6,3.6);
  \node[scale=1.3] at (0.6,0.5) {\faRobot};
  \draw[-{Triangle},line width=1.5pt,kcblue] (0.6,0.85) -- (0.6,1.15);
  \node[scale=1.4] at (3.0,3.0) {\faStar};
  \foreach \i in {1,3,4,5} {
    \begin{scope}[shift={(3.0+\i*1.75,1.0)}]
      \filldraw[edge,fill=kcblue!10] (0,0) rectangle (1.5,1.9);
      \draw[arr,kcblue] (0.75,0.25) -- (0.75,1.7);
      \node[font=\scriptsize\bfseries,align=center] at (0.75,-0.45) {FORWARD};
    \end{scope}}
  \begin{scope}[shift={(3.0+2*1.75,1.0)}]
    \filldraw[edge,fill=kcgreen!12] (0,0) rectangle (1.5,1.9);
    \draw[arr,kcgreen] (0.5,0.25) -- (0.5,1.1) arc (180:90:0.45);
    \node[font=\scriptsize\bfseries,align=center] at (0.75,-0.45) {TURN\\RIGHT};
  \end{scope}
\end{tikzpicture}
\end{center}
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Put your finger on the robot. Move it one card at a time.
  \item Circle the card in the wrong place.
  \item Fix it on the floor. Test it with the robot!
\end{enumerate}
\end{exercise}
