\chapterweek{8}{Draw With the Pen}{The cat can draw: a square is repeat 4 of move and a quarter turn.}

\section{What you will learn}
\begin{ican}
  \item I can make the cat draw with pen down.
  \item I can draw a square with repeat 4 and a quarter turn.
  \item I can fix a turn that is in the wrong place.
\end{ican}

\section{The cat has a pen}
Pen down: the cat draws when it walks. Pen up: it stops drawing. A quarter turn is 90.

\begin{grownup}[Adding the pen]
Click \menu{Add Extension} at the bottom-left. Choose \menu{Pen}. A green Pen circle appears. Do this in every new project.
\end{grownup}

\begin{exercise}[Draw a Square]
Make the cat draw a square.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Drag the cat to a place with room.
  \item Build the script. Put move and turn inside the repeat.
  \item Click the green flag. Count the sides.
\end{enumerate}
\begin{center}
\begin{scratch}[scale=0.85]
\blockinit{when \greenflag clicked}
\blocklook{set size to \ovalnum{40} \%}
\blockpen{erase all}
\blockpen{pen down}
\blockrepeat{repeat \ovalnum{4}}{
  \blockmove{move \ovalnum{100} steps}
  \blockmove{turn \turnright{} \ovalnum{90} degrees}
}
\blockpen{pen up}
\end{scratch}
\end{center}
\end{exercise}

\begin{exercise}[The One-Side Square]
Omar's cat draws one long line. It turns only once. Circle the block in the wrong place.
\begin{center}
\begin{scratch}[scale=0.85]
\blockinit{when \greenflag clicked}
\blockpen{erase all}
\blockpen{pen down}
\blockrepeat{repeat \ovalnum{4}}{
  \blockmove{move \ovalnum{100} steps}
}
\blockmove{turn \turnright{} \ovalnum{90} degrees}
\end{scratch}
\end{center}
Fix it on your Mac. Move the turn inside the repeat.
\end{exercise}

\begin{exercise}[Stamp a Ring]
Stamp leaves a picture of the cat. Make a ring of six cats.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Read the turn card. Six sides: turn 60.
  \item Build the script. Click the green flag.
\end{enumerate}
\begin{center}
\begin{scratch}[scale=0.85]
\blockinit{when \greenflag clicked}
\blocklook{set size to \ovalnum{40} \%}
\blockpen{erase all}
\blockrepeat{repeat \ovalnum{6}}{
  \blockpen{stamp}
  \blockmove{move \ovalnum{60} steps}
  \blockmove{turn \turnright{} \ovalnum{60} degrees}
}
\end{scratch}
\end{center}
\end{exercise}

\begin{vocab}
  \vocabword{pen}{A crayon for the sprite. Pen down draws. Pen up stops.}
  \vocabword{stamp}{Leaves a picture of the sprite on the stage.}
  \vocabword{quarter turn}{A turn of 90. Four quarter turns bring you back.}
\end{vocab}
