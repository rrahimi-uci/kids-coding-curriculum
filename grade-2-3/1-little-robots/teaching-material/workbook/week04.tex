\section{Repeat it!}
Scratch has a \textbf{repeat} block. It is shaped like a big C. The blocks inside it run again and again. The number says how many times. These two scripts do the same thing. Which one is shorter?

\begin{center}
\begin{minipage}[t]{0.45\linewidth}\centering
\textbf{The long way}\par\medskip
\begin{scratch}[scale=0.65]
\blockinit{when \greenflag clicked}
\blocklook{next costume}
\blockcontrol{wait \ovalnum{1} seconds}
\blocklook{next costume}
\blockcontrol{wait \ovalnum{1} seconds}
\blocklook{next costume}
\blockcontrol{wait \ovalnum{1} seconds}
\blocklook{next costume}
\blockcontrol{wait \ovalnum{1} seconds}
\end{scratch}
\end{minipage}\hfill
\begin{minipage}[t]{0.45\linewidth}\centering
\textbf{The repeat way}\par\medskip
\begin{scratch}[scale=0.65]
\blockinit{when \greenflag clicked}
\blockrepeat{repeat \ovalnum{4}}{
  \blocklook{next costume}
  \blockcontrol{wait \ovalnum{1} seconds}
}
\end{scratch}
\end{minipage}
\end{center}

\begin{exercise}[Dance, Ballerina!]
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Click Choose a Sprite. It is the cat button under the stage. Pick Ballerina.
  \item Click the Costumes tab. She has 4 dance moves! Click the Code tab.
  \item Build the repeat way. The repeat block is in orange Control.
  \item Click the green flag. Count her moves.
\end{enumerate}
\end{exercise}

\begin{exercise}[The Noisy Cat]
Leo wants his cat to meow 3 times. Click the cat first. Build Leo's script.
\begin{center}
\begin{scratch}[scale=0.9]
\blockinit{when \greenflag clicked}
\blockrepeat{repeat \ovalnum{10}}{
  \blocksound{play sound \selectmenu{Meow} until done}
}
\end{scratch}
\end{center}
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Click the green flag. Count the meows.
  \item Too noisy? Click the red stop sign.
  \item Fix the number. Test again.
\end{enumerate}
\end{exercise}

\begin{exercise}[Too Many or Too Few?]
Mei's ballerina should do all 4 moves. Here is her script.
\begin{center}
\begin{scratch}[scale=0.9]
\blockinit{when \greenflag clicked}
\blockrepeat{repeat \ovalnum{2}}{
  \blocklook{next costume}
  \blockcontrol{wait \ovalnum{1} seconds}
}
\end{scratch}
\end{center}
Circle one: \qquad \textbf{too many} \qquad \textbf{too few} \qquad \textbf{right}
\end{exercise}
