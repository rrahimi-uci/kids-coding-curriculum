\begin{core}[Loot Bag (Thonny, 10 minutes)]
Open \texttt{exercises/week03/hw-loot-bag/starter.py}. You open a treasure chest several times and keep the loot in a list.
\begin{enumerate}[leftmargin=1.4em]
  \item Press \menu{Run}. The bag is empty!
  \item TODO 1: in \texttt{fill\_bag()}, repeat \texttt{how\_many} times and \texttt{append} \texttt{open\_chest(chest)} to the bag.
  \item TODO 2: in \texttt{count\_item()}, add 1 to \texttt{count} each time a thing in the bag matches the item.
  \item Run it three times. Is the loot the same each time? Why not? \rule{6cm}{0.4pt}
  \item Open \texttt{check\_my\_work.py} and press \menu{Run}.
\end{enumerate}
\end{core}

\begin{challenge}[Rare treasure]
\texttt{"gold coin"} is in \texttt{CHEST} three times, so it is three times as likely as the ruby. Add a super-rare item that
appears only once, and make \texttt{"old boot"} more common. Then open the chest 20 times: how many of your rare item did you get?
\rule{2cm}{0.4pt}
\end{challenge}

\begin{creative}[Design a palette]
Choose 4 colors from the Okabe-Ito list for a game or poster you would like to make. Write the list as Python, then sketch
your design. Add a label or shape to every colored part, so someone who sees colors differently can still understand it.
\answerlines{1}
\answerspace{3.5cm}
\end{creative}
