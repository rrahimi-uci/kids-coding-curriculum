\begin{challenge}[Mode detective]
Predict what is in \texttt{test.txt} after each program runs (start with no file each time). Then try it in Thonny.
\begin{pycode}[Program A]
for word in ["red", "blue"]:
    with open("test.txt", "w") as file:
        file.write(word + "\n")
\end{pycode}
\begin{pycode}[Program B]
for word in ["red", "blue"]:
    with open("test.txt", "a") as file:
        file.write(word + "\n")
\end{pycode}
Program A gives: \rule{4cm}{0.4pt} \qquad Program B gives: \rule{4cm}{0.4pt}
\par Then add \texttt{d} = delete all ideas to your notebook (hint: which mode empties a file?).
\end{challenge}
