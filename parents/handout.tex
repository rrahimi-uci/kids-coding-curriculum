\section{What the research says}
\begin{itemize}
  \item \evS\enspace Children learn to program; problem solving and planning improve
    \parencite{scherer2019transfer,montuori2024ct}.
  \item \evM\enspace Math and creativity gains shrink in stricter studies; reading and grades
    benefit least \parencite{scherer2019transfer}.
  \item \evM\enspace Blocks before text: no lasting harm or edge \parencite{xu2019blocks}.
  \item \evO\enspace General persistence; long-term effects of AI \parencite{kazemitabaar2023codex}.
\end{itemize}
\textbf{Our promise:} capable, confident beginner programmers. No promises about grades, jobs or college.

\section{How we teach, and why}
Real projects in games and art \parencite{resnick2009scratch}; pair programming
\parencite{umapathy2017pair}; a bug hunt every week \parencite{michaeli2019debugging}; unplugged
and predict-then-run activities \parencite{sentance2019primm}; ideas that return, deeper, each level.
