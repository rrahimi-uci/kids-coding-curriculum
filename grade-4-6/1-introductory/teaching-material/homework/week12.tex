\begin{challenge}[Holiday \enquote{keep coding} menu: pick 2]
No internet or account needed. Save each one in \texttt{Documents/my-python}.
\begin{enumerate}[leftmargin=1.6em,itemsep=2pt]
  \item \textbf{Magic spiral}: \py|for i in range(60):| with \py|t.forward(i * 3)| and \py|t.left(91)|. Try other turns.
  \item \textbf{Kaleidoscope}: 36 squares, turning \py|t.left(10)| after each, with \py|random.choice(PALETTE)| colors.
  \item \textbf{Dice roller}: \py|roll = random.randint(1, 6)|; print it; if it is 6, print \enquote{Roll again!}.
  \item \textbf{City skyline}: a \texttt{def building(height, color):} that draws a filled rectangle, called 12 times with random heights.
  \item \textbf{Times-table quiz}: your Mini Math Quiz with 5 questions and a message for each score.
  \item \textbf{Remix your art show}: add a new function, or a second question.
  \item \textbf{Unplugged}: teach someone at home the \enquote{parent-robot} game from week 1 --- you give the instructions!
\end{enumerate}
