% Level 3 homework, Week 11 -- Project Studio 2
\homeworkweek{11}{Project Studio 2}{15--20 minutes. Finish Milestone 4 and get ready for the showcase.}
\namedateline

\begin{core}[Finish Milestone 4 and prepare your talk]
\begin{enumerate}[itemsep=1pt]
  \item Finish your \texttt{README.md}: title, how to run, controls, how it works (classes, data structure, algorithm, saved file), tests, credits. Commit it.
  \item Fix one more \enquote{must fix} or \enquote{should fix} bug from your playtest cards. Commit it.
  \item Run all your tests: \texttt{python -m pytest -q}. Number passing: \rule{1.5cm}{0.4pt} (goal: 5 or more)
  \item Fill in the showcase talk card (workbook Exercise 12.1 -- the page handed out today) and practice once out loud with a timer: 2 minutes.
\end{enumerate}
\end{core}

\begin{challenge}[An edge-case hunt]
Think like a sneaky player: what could break your project? (Pressing keys very fast, walking into a corner, an empty save file, the last life.) Write a test for one of them. If it fails, fix the code. Commit with a message like \texttt{Test and fix empty high-score file}.
\answerlines{2}
\end{challenge}

\begin{creative}[Box art]
Design the box or poster for your project as if it were sold in a shop: the title, one exciting sentence, a picture, and \enquote{Made by} with your \emph{nickname}. Keep it at home -- no posting online.
\answerspace{4cm}
\end{creative}

\begin{selfcheck}
  \item I have 5 or more passing tests, including an edge case.
  \item My README explains my classes, data structure and algorithm.
  \item My project passes the accessibility checklist (not color alone, readable text, controls on screen).
  \item I have 6 or more commits and know my talk's Problem, Choice, Why and Next time.
\end{selfcheck}
