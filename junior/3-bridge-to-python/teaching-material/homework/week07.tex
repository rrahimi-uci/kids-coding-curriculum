\homeworkweek{7}{Repeat in Python}{Optional. One short game with a grown-up.}

\begin{playathome}[The indent dance]
Here is a dance program. Lines that are pushed in are inside the repeat.
\begin{center}
\begin{tabular}{@{}l@{}}
\texttt{for i in range(3):}\\
\texttt{\ \ \ \ clap}\\
\texttt{\ \ \ \ stomp}\\
\texttt{wave}
\end{tabular}
\end{center}
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Your grown-up reads it. You dance it. How many waves? (Count them!)
  \item Now push \texttt{wave} in, under \texttt{stomp}. Dance it again. How many waves now?
  \item Write your own dance with a grown-up. Swap: you read, they dance.
\end{enumerate}
\end{playathome}

\begin{playathome}[If you have Thonny at home]
Open your flower. Change \texttt{80} to \texttt{50}. Run it. A smaller flower!
\end{playathome}

\begin{selfcheck}
  \item I know which lines are inside the repeat.
  \item I can say what \texttt{for i in range(3):} means.
\end{selfcheck}
