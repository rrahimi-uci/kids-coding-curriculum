% Junior Level 3, Week 2 -- play-at-home card
\homeworkweek{2}{Plan It on Paper}{Optional. One short game with a grown-up.}

\begin{playathome}[Plan a picnic in parts]
Plan a picnic (real or pretend) with your grown-up.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Draw three boxes. Name the parts: FOOD, GAMES, PACKING.
  \item Write or draw two steps in each box.
  \item Give each part a job holder: you, your grown-up, a toy.
  \item Check: do the parts fit together? If the food is for 4 people, do you pack 4 cups?
\end{enumerate}
\end{playathome}

\begin{playathome}[Get ready for Data Detectives]
Next week you run a survey. Ask your grown-up to help you find people to answer one question: \enquote{What is your favorite fruit?} Toys can vote too!
\end{playathome}

\begin{selfcheck}
  \item I planned something in parts.
  \item I can say how to test one part.
\end{selfcheck}
