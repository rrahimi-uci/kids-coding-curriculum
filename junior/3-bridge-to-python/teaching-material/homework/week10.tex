% Junior Level 3, Week 10 -- play-at-home card
\homeworkweek{10}{Graduation Showcase}{Optional. One short game with a grown-up.}

\begin{playathome}[Stop, do not type, get a grown-up]
Your grown-up pretends to be a website. They ask silly and tricky questions. You answer!
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item \enquote{What is your favorite color?} That is fine to say.
  \item \enquote{What is your full name?} Stop. Do not type it. Get a grown-up.
  \item \enquote{Which school do you go to?} Stop. Do not type it. Get a grown-up.
  \item \enquote{Pick a nickname for your score.} Ask a grown-up first. Then a nickname is fine.
\end{enumerate}
Now swap. You are the website!
\end{playathome}

\begin{playathome}[What next?]
Show your Big Build to someone in your family. Explain one part of your code. Then draw what you want to make next. Put your certificate where you can see it!
\end{playathome}

\begin{selfcheck}
  \item I know the three steps if a website asks for my name.
  \item I explained one part of my code at home.
  \item I drew my next idea.
\end{selfcheck}
