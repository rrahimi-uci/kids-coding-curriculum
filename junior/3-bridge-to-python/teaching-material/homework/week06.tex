% Junior Level 3, Week 6 -- play-at-home card
\homeworkweek{6}{Hello, Python!}{Optional. One short game with a grown-up.}

\begin{playathome}[The picky robot]
You are a robot that knows only three words: \textbf{forward}, \textbf{left} and \textbf{right}.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Your grown-up writes commands on paper, one per line, like \texttt{forward 3} and \texttt{left}.
  \item One command has a typo, like \texttt{forwad 2}.
  \item Do each line, top to bottom. At the typo, freeze! Say: \enquote{Error on line 4!}
  \item Fix the typo with a pencil. Start again from the top.
  \item Swap: you write, your grown-up is the robot.
\end{enumerate}
\end{playathome}

\begin{playathome}[If you have Thonny at home]
Open your square. Change \texttt{100} to \texttt{50} in all four lines. Run it. A tiny square!
\end{playathome}

\begin{selfcheck}
  \item I found a typo and said its line number.
  \item I can say what \texttt{t.forward(100)} does.
\end{selfcheck}
