\begin{exercise}[Color My Square]
Folder: \texttt{week08/ex-color-square}.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Load \texttt{starter.py}. Run it. A thin black square.
  \item Change \texttt{black} to a color you like. Keep the quote marks!
  \item Save and Run.
  \item Change \texttt{t.pensize(1)} to \texttt{t.pensize(5)}. Save and Run.
\end{enumerate}
\textbf{Bug hunt:} one color has a typo. Python shows red text: \texttt{bad color string}. Circle the typo.
\begin{center}
\texttt{t.color("gold")}\qquad\texttt{t.color("bleu")}\qquad\texttt{t.color("green")}
\end{center}
\end{exercise}

\begin{exercise}[Two-Color Flower]

\end{exercise}

\begin{exercise}[My Square Block (stretch)]
Folder: \texttt{week08/ex-square-block}. Python has My Blocks too!
\begin{center}
\begin{scratch}[scale=0.8]
\blockmoreblocks{define \ovalmoreblocks{square}}

  \blockmove{move \ovalnum{60} steps}
  \blockmove{turn \turnleft{} \ovalnum{90} degrees}
}
\end{scratch}
\end{center}
\begin{pycode}
def square():
    for side in range(4):
        t.forward(60)
        t.left(90)
\end{pycode}
Change \texttt{range(1)} at the bottom to \texttt{range(4)}. Save and Run. A window!
\end{exercise}
