  \item I can fix a typo in Python.

\end{ican}

\begin{exercise}[Last-Minute Bug Check]
Check your Big Build with a partner. Circle yes or not yet.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item It starts. \hfill yes \quad not yet
  \item It starts the same way the second time. \hfill yes \quad not yet
  \item It has every must-have. \hfill yes \quad not yet
  \item Everyone can see it. \hfill yes \quad not yet
\end{enumerate}
Fix one thing. Then save.

\textbf{Typo rescue!} Open \texttt{typo\_rescue.py}. Press Run. Read the error message. Find the line. Fix the typo. Run again. Then change one color and one number.
\end{exercise}

\begin{exercise}[My Showcase]
Show your Big Build. Then say three sentences.

\noindent\begin{minipage}{\linewidth}
\begin{pycode}
for i in range(3):
    t.forward(100)
    t.left(120)
\end{pycode}
\end{minipage}
Omar says: \enquote{This loop draws my roof. It goes three times, because a triangle has three sides.}

Now you:
\begin{itemize}[leftmargin=1.4em,itemsep=2pt]
  \item My plan card said \rule{6cm}{0.4pt}.
  \item This part of my code \rule{5.5cm}{0.4pt}.
  \item I changed \rule{4cm}{0.4pt}. My idea came from \rule{3cm}{0.4pt}.
\end{itemize}
\end{exercise}
