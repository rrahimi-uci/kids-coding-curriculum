% Junior Level 2, Week 3 -- play-at-home card
\homeworkweek{3}{Keep Score}{Optional. One short game with a grown-up.}

\begin{playathome}[Sock ball with score boxes]
You need two cups, some buttons or beans, a sock ball and a basket. Label the cups SCORE and LIVES.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Say the rules out loud first. Then set SCORE to 0 (empty it). Set LIVES to 3 (put in 3 buttons).
  \item Throw the sock ball. In the basket? Change SCORE by 1: add a button.
  \item Missed? Take one button out of LIVES.
  \item No lives left? Count your score. Now it is the grown-up's turn. What do you do first?
\end{enumerate}
\end{playathome}

\begin{selfcheck}
  \item I said the rules before we played.
  \item I set the score to 0 before each turn.
  \item I can say what a variable is.
\end{selfcheck}
