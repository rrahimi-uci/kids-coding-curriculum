% Junior Level 2, Week 2 -- play-at-home card
\homeworkweek{2}{If Touching\ldots}{Optional. One short game with a grown-up.}

\begin{playathome}[Simon says IF]
Take turns with a grown-up. Say IF and a rule.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item \enquote{If you have a sock on, jump!}
  \item \enquote{If you like pizza, touch your nose!}
  \item Check yourself. True? Do it. Not true? Stay still.
\end{enumerate}
Now a forever rule: \enquote{Again and again: if I hold up a spoon, clap!} Keep checking!
\end{playathome}

\begin{playathome}[Two stars and a wish]
Ask your grown-up to make a drawing or a block tower. Give two stars and a wish: two things you like, and one kind idea to make it even better.
\end{playathome}

\begin{selfcheck}
  \item I played Simon says IF.
  \item I can say what an if block does.
  \item I gave two stars and a wish.
\end{selfcheck}
