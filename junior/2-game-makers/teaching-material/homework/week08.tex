% Junior Level 2, Week 8 -- play-at-home card
\homeworkweek{8}{Start, Win and Lose}{Optional. One short game with a grown-up.}

\begin{playathome}[Rule detectives]
Play a short game with a grown-up: cards, a board game, or tic-tac-toe. Then answer together:
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item How does it start?
  \item How do you win?
  \item How do you lose, or how does it end?
\end{enumerate}
Does the game always end? Good games do!
\end{playathome}

\begin{playathome}[My arcade game idea]
Next week you build your own arcade game. Draw your idea: the player, the thing to catch or the maze, and the stage. Tell a grown-up how you win and how you lose. Bring the drawing next week.
\answerspace{3cm}
\end{playathome}

\begin{selfcheck}
  \item I can name the three parts of a game: start, play, ending.
  \item My game says You win! or Game over! and stops.
  \item I can stop playing when a game says \enquote{just one more}.
\end{selfcheck}
