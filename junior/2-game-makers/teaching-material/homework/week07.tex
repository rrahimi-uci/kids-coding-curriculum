% Junior Level 2, Week 7 -- play-at-home card
\homeworkweek{7}{My Own Blocks}{Optional. One short game with a grown-up.}

\begin{playathome}[My named routine]
Pick a job you do every day, like getting ready for bed.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Draw its steps on small papers, one step on each.
  \item Put them in an envelope. Write a name on it, like \enquote{BEDTIME}.
  \item The grown-up says the name. You do every step inside!
  \item Swap: you say the name, the grown-up does the steps.
\end{enumerate}
\end{playathome}

\begin{playathome}[Shape detectives]
Look at a board game, a card game or a game on a box. Can you tell the pieces apart by shape, not only by color? Find one thing that works for everyone, and one thing that could be better.
\end{playathome}

\begin{selfcheck}
  \item I made a named routine with steps inside.
  \item I know a My Block must be used, or it does nothing.
  \item I can say one way to make a game work for everyone.
\end{selfcheck}
