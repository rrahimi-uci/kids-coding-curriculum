% Junior Level 2, Week 3 -- workbook chapter (My Game Book)
\chapterweek{3}{Keep Score}{A variable is a labeled box that remembers a number.}

\section{What you will learn}
\begin{ican}
  \item I can make a score variable.
  \item I can set the score to 0 and change it by 1.
  \item I can fix a score that does not go back to 0.
\end{ican}

\section{A box for a number}
A \textbf{variable} is a box with a label. It remembers a number. Variable blocks are orange-red.

\begin{center}

\begin{exercise}[Pop the Balloon]
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Add Balloon1.
  \item Orange-red \menu{Variables}: Make a Variable. Type \enquote{score}. Pick For all sprites.
  \item Build both scripts.
  \item Click the flag. Pop the balloon five times. Look at the score.
  \item Click the flag again. What happens to the score?
\end{enumerate}
\begin{center}
\begin{scratch}[scale=0.8]
\blockinit{when \greenflag clicked}
\blockvariable{set \selectmenu{score} to \ovalnum{0}}
\blockvariable{show variable \selectmenu{score}}
\end{scratch}
\qquad
\begin{scratch}[scale=0.8]
\blockinit{when this sprite clicked}
\blockvariable{change \selectmenu{score} by \ovalnum{1}}
\blocksound{start sound \selectmenu{Pop}}
\blocklook{next costume}
\end{scratch}
\end{center}
\end{exercise}

\begin{exercise}[Catch and Count]
Open your catch game from last week.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Make a score variable, for all sprites.
  \item Click the Apple. Add the two orange-red blocks.
  \item Play! Each catch is 1 point.

\begin{center}
\begin{scratch}[scale=0.7]
\blockinit{when \greenflag clicked}
\blockvariable{set \selectmenu{score} to \ovalnum{0}}
\blockmove{set rotation style \selectmenu{left-right}}
\blockmove{go to \selectmenu{Star}}
\blockmove{point in direction \ovalnum{180}}
\blockinfloop{forever}{
  \blockmove{move \ovalnum{5} steps}
  \blockif{if \boolsensing{touching \ovalsensing*{Bowl} ?} then}{
    \blockvariable{change \selectmenu{score} by \ovalnum{1}}
    \blocksound{start sound \selectmenu{Pop}}
    \blockmove{go to \selectmenu{Star}}
  }
  \blockif{if \boolsensing{touching \ovalsensing*{edge} ?} then}{
    \blockmove{go to \selectmenu{Star}}
  }
}
\end{scratch}
\end{center}
\end{exercise}

\begin{vocab}
  \vocabword{variable}{A box with a label that remembers a number, like score.}
  \vocabword{set}{Put a new number in the box.}
  \vocabword{change}{Add to the number that is already in the box.}
\end{vocab}
