% Junior Level 1, Week 4 -- play-at-home card
\homeworkweek{4}{Repeat It!}{Optional. One short game with a grown-up.}

\begin{playathome}[Dance loops at home]
Make dance cards with a grown-up: STOMP, CLAP, JUMP, WIGGLE.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Pick two cards for your dance. Put a loop of string around them.
  \item Write a REPEAT card with a number from 2 to 5.
  \item Teach your dance to your grown-up. Dance it together!
  \item Now your grown-up dances it with a bug on purpose. Too many or too few?
\end{enumerate}
\end{playathome}

\begin{playathome}[Repeat hunt]
Look for repeats at home. A song with a part that comes again? Stairs: step, step, step? Tell your grown-up one repeat you found, and how many times it repeats.
\end{playathome}

\begin{selfcheck}
  \item I made a dance with a repeat card.
  \item I can say how many times a repeat runs.
  \item I took turns fairly.
\end{selfcheck}
