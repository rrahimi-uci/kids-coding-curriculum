% Junior Level 1, Week 10 -- play-at-home card
\homeworkweek{10}{Showcase and Certificate}{Optional. One short game with a grown-up.}

\begin{playathome}[Show your card at home]
Show your greeting card to someone in your family.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Click the green flag. Show the surprise.
  \item Point at one script. Tell them how it works: \enquote{When you click the \ldots, it \ldots}
  \item Ask them: \enquote{Which part did you like best?}
\end{enumerate}
\end{playathome}

\begin{playathome}[What next?]
Draw one thing you want to make next in Scratch. A game? A story? A drawing with the pen? Put your certificate somewhere you can see it!
\answerspace{3cm}
\end{playathome}

\begin{selfcheck}
  \item I showed my card to my family.
  \item I said how one script works.
  \item I used my nickname, not my full name.
\end{selfcheck}
