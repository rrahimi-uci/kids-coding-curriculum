% Junior Level 1, Week 1 -- play-at-home card
\homeworkweek{1}{Robots Need Instructions}{Optional. One short game with a grown-up.}

\begin{playathome}[Robot in the kitchen]
A grown-up is the robot. You are the coder.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Pick a goal: the fridge, the door or a toy.
  \item Say one step at a time: FORWARD, TURN LEFT or TURN RIGHT.
  \item FORWARD is one big step. A turn is a quarter turn, on the spot.
  \item The robot does exactly what you say. Did it get there?
  \item Swap: now you are the robot!
\end{enumerate}
\end{playathome}

\begin{playathome}[Ask a grown-up]
Sit with a grown-up and look at a game or app you use. Can you find a place where it asks for something, like a name, a photo or a tap on OK? Show what you do: stop, and ask a grown-up.
\end{playathome}

\begin{selfcheck}
  \item I gave a robot exact steps, one at a time.
  \item I know my coding nickname.
  \item When an app asks for something, I ask a grown-up.
\end{selfcheck}
