\section{What you will learn}

  \item I can say what one script does.
  \item I can give kind sticker feedback.
\end{ican}

\section{Showcase day}
A \textbf{showcase} is a little show. You show your card. You say what your code does. Then we celebrate!

\begin{exercise}[Last-Minute Bug Check]
Check your card with a partner. Circle yes or not yet.
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Click the green flag. Does it start? \hfill yes \quad not yet
  \item Click the flag again. Same start? \hfill yes \quad not yet
  \item Click or press the key. Surprise? \hfill yes \quad not yet
  \item Can you hear the sound? \hfill yes \quad not yet
\end{enumerate}
Fix one thing. Then save.
\end{exercise}

\begin{exercise}[My Showcase]
Show your card. Then point at one script. Say it from the top.
\begin{center}
\begin{scratch}[scale=0.85]
\blockinit{when this sprite clicked}
\blocksound{play sound \selectmenu{Pop} until done}
\blocklook{say \ovalnum{Surprise!} for \ovalnum{2} seconds}
\end{scratch}
\end{center}
Mei says: \enquote{When you click the balloon, it pops and says Surprise!}

Now you:
\begin{itemize}[leftmargin=1.4em,itemsep=2pt]
  \item My card is for \rule{4cm}{0.4pt}.
  \item When you \rule{3.5cm}{0.4pt}, it \rule{4cm}{0.4pt}.
\end{itemize}
\end{exercise}

\begin{exercise}[Sticker Feedback]
Visit a friend's card. Ask before you click. Put a sticker on the part you liked best. Say it out loud!
\begin{center}

  lab/.style={font=\bfseries,align=center}]
  \foreach \i/\t/\ic in {0/dance/\faMusic,1/sound/\faVolumeUp,2/surprise/\faGift} {

\end{center}
\end{exercise}
